  - lr = proto.stats.plasticity × policy.hebbian_rate (Step 7's responsibility)

  /// What Step 4 routing proposes to do to the substrate. Held through
  /// the read-mostly phase, applied by Step 7 (`apply_region_delta`)
  /// during the mutation phase. Verbatim from §10.2 of `new_foundation.md`.
  #[derive(Default, Debug, Clone)]
  pub struct RegionDelta {
      /// `(child, parent, weight)` — committed as new
      /// `(child, parent_region, parent)` relations with
      /// `stats.confidence = weight`, or as confidence reinforcement on
      /// existing parent_region relations.
      pub parent_attachments: Vec<(ElementId, ElementId, f32)>,

      /// `(prototype_element, target_embedding)` — committed by
      /// overwriting the prototype Element's inline `embedding` via
      /// spherical k-means: `new = normalize(old + lr · (target − old))`
      /// where `lr = proto.stats.plasticity * policy.hebbian_rate`.
      /// Step 4 stores only the target; Step 7 reads policy + plasticity
      /// and does the math.
      pub prototype_updates: Vec<(ElementId, Vec<f32>)>,

      /// Empty in v0 Step 4 — mid-path insertion (§10.3.5) needs span-
      /// level embeddings, which only exist after Step 6 mints elements.
      /// Step 8 owns this list. Field exists so the struct stays
      /// stable across pipeline phases.
      pub new_regions: Vec<NewRegion>,

      /// Empty in v0 Step 4 for the same reason — authoritative member
      /// assignment uses each minted element's own embedding (Step 8).
      pub new_members: Vec<(ElementId, ElementId)>,

      /// Number of routing branches that died at sub-threshold quality
      /// (best child below `policy.leaf_vigilance`). Surfaces in the
      /// frame as a routing-quality signal.
      pub void_count: u32,
  }

  /// Mid-path insertion candidate. Populated by Step 8 from span-level
  /// embeddings; written by Step 7 as a Defeasible `parent_region`
  /// relation pending replay confirmation. Not used in v0 Step 4.
  #[derive(Debug, Clone)]
  pub struct NewRegion {

  use crate::math::dot;
  use crate::types::{
      ElementId, Hypergraph, Policy, RegionActivation, RegionDelta, UncertaintySignal,
  };
  use std::collections::HashSet;

  /// What Step 4 returns to the rest of the read-mostly phase. The
  /// `delta` is held through Steps 5–6 and committed by Step 7; the
  /// active region list seeds Step 5's warm-bias label set; the
  /// `uncertainty` signals get appended to the per-tick buffer Step 13
  /// collects.
  pub struct RouteResult {
      pub active_regions: Vec<RegionActivation>,
      pub delta: RegionDelta,
      pub uncertainty: Vec<UncertaintySignal>,
  }

  /// Step 4 of the tick pipeline. Read-only DAG descent from GENESIS,
  /// scoring each candidate region's prototypes against the input
  /// embedding and proposing structural updates to the routed-through
  /// path. Pure: no model calls, no allocation beyond the result.
  ///
  /// `embedding` must be the L2-normalized sentence embedding produced
  /// by `embed::embed_text` — cosine similarity is computed as raw dot
  /// product, which is wrong for non-unit vectors.
  ///
  /// Two-stage check at each node:
  /// - If the best child's score < `policy.leaf_vigilance`, the
  ///   current branch dies into VOID (`delta.void_count += 1`); descent
  ///   stops.
  /// - Otherwise, descend into every child whose score
  ///   ≥ `policy.descend_threshold` (no K cap; the prototype set per
  ///   region is kept small by the merge/split policies).
  ///
  /// Children whose score ≥ `policy.region_activation_threshold` land
  /// in `active_regions`. If the final active set is empty,
  /// `DiffuseRouting` is raised so Step 13 can surface it.
  ///
  /// Mid-path insertion (§10.3.5) is deliberately skipped here — that
  /// pass uses span-level embeddings that don't exist until Step 6
  /// mints elements. Step 8 owns it.
  pub fn route_regions(
      embedding: &[f32],
      hg: &Hypergraph,
      policy: &Policy,
  ) -> RouteResult {
      let mut active_regions: Vec<RegionActivation> = Vec::new();
      let mut delta = RegionDelta::default();
      let mut uncertainty: Vec<UncertaintySignal> = Vec::new();

      let mut visited: HashSet<ElementId> = HashSet::new();
      let mut frontier: Vec<ElementId> = vec![hg.genesis];

      while !frontier.is_empty() {
          let mut next_frontier: Vec<ElementId> = Vec::new();
          for current in frontier {
              if !visited.insert(current) {
                  continue;
              }

              // Structural leaf — no children to score.
              let Some(children) = hg.region_children.get(&current) else {
                  continue;
              };
              if children.is_empty() {
                  continue;
              }

              // Score each child by max cosine over its prototypes.
              // Tracks (child_id, best_score, best_proto_id).
              let mut scored: Vec<(ElementId, f32, ElementId)> =
                  Vec::with_capacity(children.len());
              for &child in children {
                  let Some(protos) = hg.region_prototypes.get(&child) else {
                      continue;
                  };
                  let mut best: Option<(ElementId, f32)> = None;
                  for &proto_id in protos {
                      let proto = &hg.elements[proto_id.0 as usize];
                      let sim = dot(embedding, &proto.embedding);
                      if best.map_or(true, |(_, s)| sim > s) {
                          best = Some((proto_id, sim));
                      }
                  }
                  if let Some((p, s)) = best {
                      scored.push((child, s, p));
                  }
              }
              if scored.is_empty() {
                  continue;
              }

              // Sub-threshold leaf check — best child below the quality
              // floor → this branch routes to VOID and we don't descend.
              let best_score = scored
                  .iter()
                  .map(|(_, s, _)| *s)
                  .fold(f32::NEG_INFINITY, f32::max);
              if best_score < policy.leaf_vigilance {
                  delta.void_count += 1;
                  continue;
              }

              // Descend into every child clearing descend_threshold.
              for (child, score, best_proto) in scored {
                  if score < policy.descend_threshold {
                      continue;
                  }
                  next_frontier.push(child);
                  delta.parent_attachments.push((child, current, score));
                  delta.prototype_updates.push((best_proto, embedding.to_vec()));
                  if score >= policy.region_activation_threshold {
                      active_regions.push(RegionActivation {
                          region: child,
                          activation: score,
                      });
                  }
              }
          }
          frontier = next_frontier;
      }

      if active_regions.is_empty() {
          uncertainty.push(UncertaintySignal::DiffuseRouting);
      }

      RouteResult {
          active_regions,
          delta,
          uncertainty,
      }
  }

  Where RouteResult lives: local to this file. It's a one-call-site struct just to bundle three return values. Doesn't
  belong in types.rs since nothing else references it (Step 7 only takes RegionDelta, Step 13 only takes
  Vec<RegionActivation> from the frame).

  Embedding cloning: ~15 clones × 1.5 KB = ~23 KB per tick. Insignificant. We could share a single owned Vec<f32> and
  have updates store indices into it, but that couples to v0 semantics ("target is always the input embedding"); when
  Step 8 lands span-level updates, targets vary per prototype. Match the spec's (proto, embedding) pair shape verbatim.
